    if value <= 10_000:
        k_min, k_max, round_to = 0.70, 1.30, 1

    k = random.uniform(k_min, k_max)

class ClipboardHandler:
    def is_table_like(self, text):
        lines = text.strip().split("\n")
        return len(lines) > 1 and any("\t" in line for line in lines)

    def process_clipboard_content(self, content):
        if not self.is_table_like(content):
            return content
        return self.mask_table_data(content)

    def start_monitoring(self):
        while self.is_running:
            current = pyperclip.paste()
            if current != self.last_clipboard_content:
                processed = self.process_clipboard_content(current)
                if processed != current:
                    pyperclip.copy(processed)
            time.sleep(self.polling_interval / 1000.0)
```

Тезисно по вариантам anti-loop:
- Хэш-контроль: сравнение MD5/SHA текущего содержимого с предыдущим, при совпадении блокируем повторную обработку.
- Счетчик повторов: ведем счетчик по хэшу, при превышении порога (например, 3-5) пропускаем обработку.
- Временной контроль: если изменения происходят слишком быстро (например, <100-200 мс), считаем цикл.
- Флаг "обработано": добавляем маркер/метку, при обнаружении свежей метки пропускаем.
- История изменений: анализ последних N состояний, выявление цикличных паттернов.
- Смешанная стратегия: комбинация методов, если срабатывает любой - блокируем повтор.

Гипотезы:
- Частота polling и скорость записи в буфер сопоставимы, поэтому возможна бесконечная переработка.

class ClipboardHandler:
    def __init__(self, polling_interval=1000):
        self.last_clipboard_hash = None

    def calculate_content_hash(self, content):
        return hashlib.md5(content.encode("utf-8")).hexdigest()

    def should_process_content(self, current_content):
        current_hash = self.calculate_content_hash(current_content)
        if current_hash == self.last_clipboard_hash:
            return False
        self.last_clipboard_hash = current_hash
        return True
```
